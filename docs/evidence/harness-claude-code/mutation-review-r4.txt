import os, shutil, difflib, hashlib, sys
R = r"C:\repos\Knowledge-engine"
OUT = sys.argv[1]
B = chr(92)
H = os.path.join(R, "scripts", "hooks", "block-main-commit.sh")
orig = open(H, "rb").read().decode()
muts = {
 "a-drop-lowercase": (" | tr '[:upper:]' '[:lower:]'", ""),
 "b-drop-backslash-delete": ('tr -d "' + B + '"' + "'" + B*4 + '"', 'tr -d "' + B + '"' + "'" + '"'),
 "c-drop-main-master": ("((main|master)($W|" + B + "$)|--mirror|--all)", "(--mirror|--all)"),
 "d-drop-on-main-rule": ('if [ "$branch" = "main" ] || [ "$branch" = "master" ]; then', "if false; then"),
 "e-drop-continuation-join": ("sed -z 's/" + B*3 + "n/ /g' | ", ""),
 "f-drop-short-f-cluster": ("|(.*[[:space:]])?-[[:alpha:]]*f", ""),
 "g-drop-plus-refspec": ("|(.*[[:space:]:])?" + B*2 + "+", ""),
 "h-drop-force-long": ("(.*$W)?--force|", ""),
 "i-break-jq-filter": (".tool_input.command //", ".tool_input.commandx //"),
 "j-head-to-tail": ("head -n 1", "tail -n 1"),
 "k-drop-json-newline": (" -e 's/" + B*2 + "n/" + B + "n/g'", ""),
 "l-drop-empty-exit": ('[ -n "$text" ] || exit 0', ":"),
 "m-drop-am": ("|am|", "|"),
 "n-drop-mirror-all": ("|--mirror|--all", ""),
}
os.makedirs(OUT, exist_ok=True)
with open(os.path.join(OUT, "diffs.txt"), "w", newline="\n") as dl:
    for name, (old, new) in [("base", ("", ""))] + list(muts.items()):
        d = os.path.join(OUT, name)
        shutil.rmtree(d, ignore_errors=True)
        os.makedirs(os.path.join(d, "scripts", "hooks")); os.makedirs(os.path.join(d, "tests")); os.makedirs(os.path.join(d, "docker"))
        if name != "base":
            assert orig.count(old) == 1, name + " target count %d" % orig.count(old)
            m = orig.replace(old, new, 1)
        else:
            m = orig
        open(os.path.join(d, "scripts", "hooks", "block-main-commit.sh"), "w", newline="\n").write(m)
        shutil.copy(os.path.join(R, "tests", "test_block_main_commit_hook.py"), os.path.join(d, "tests"))
        open(os.path.join(d, "tests", "__init__.py"), "w").close()
        shutil.copy(os.path.join(R, "docker", "gates.Dockerfile"), os.path.join(d, "docker"))
        diff = "".join(difflib.unified_diff(orig.splitlines(True), m.splitlines(True), "hook", name, n=0))
        dl.write("=== " + name + "\n" + (diff or "(identical to live hook)\n"))
