## tmk.py
import json, os
os.makedirs("t", exist_ok=True)
G = "gi" + "t"; N = 1_000_000; NL = chr(10); BS = chr(92); DQ = chr(34)
shapes = {
 "unquoted-token": "echo " + "a" * N,
 "m-message": G + " commit -m " + DQ + "a" * N + DQ,
 "heredoc-line": "cat <<'EOF'" + NL + "a" * N + NL + "EOF",
 "many-segments": "echo x && " * 100_000 + "echo x",
 "push-many-W": G + " push " + "a/" * (N // 2),
 "push-repeated": (G + " push origin x ") * (N // 19),
 "escaped-quotes": "echo " + DQ + (BS + DQ) * (N // 2) + DQ,
 "backslashes": "echo " + BS * N,
 "push-lines": (G + " push origin agent/x" + NL) * (N // 24),
 "push-dashes": G + " push " + "-" * N,
 "git-words": (G + " ") * (N // 4) + "push",
 "continuations": ("a " + BS + NL) * (N // 4),
}
with open("t/list", "w", newline=NL) as f:
    for k, c in shapes.items():
        p = json.dumps({"session_id":"s","hook_event_name":"PreToolUse","tool_name":"Bash","tool_input":{"command":c,"description":"d"}}, separators=(",",":"))
        open("t/" + k + ".json", "w", newline=NL).write(p)
        f.write(k + NL)

## time.sh
#!/usr/bin/env bash
HOOK=$1; W=$2; gc=gi; gc=${gc}t
mkdir -p $W/fake; printf '#!/bin/sh\nexit 127\n' > $W/fake/jq; chmod +x $W/fake/jq
rm -rf $W/r; $gc init -q -b main $W/r
for br in agent/x main; do
  $gc -C $W/r symbolic-ref HEAD refs/heads/$br
  for m in jq nojq; do P=$PATH; [ $m = nojq ] && P="$W/fake:$PATH"
    while read -r k; do
      s=${EPOCHREALTIME/./}; out=$(cd $W/r && PATH="$P" bash "$HOOK" < /s/t/$k.json); rc=$?; e=${EPOCHREALTIME/./}
      case "$out" in *deny*) d=deny;; '') d=silent;; *) d=other;; esac
      printf '%-8s %-5s %-16s rc=%s %-6s %s\n' "$br" "$m" "$k" "$rc" "$d" "$(( (e - s) / 1000 ))ms"
    done < /s/t/list
  done
done
