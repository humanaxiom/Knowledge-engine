# usage: mutate.py <base> <out> ; writes one dir per probe mutant under <out>

from pathlib import Path
base, out = Path(sys.argv[1]), Path(sys.argv[2])

M = {
 "R1 field: no -f check": [('  local fl_k="" fl_v="" fl_n=0\n  [ -f "$1" ] || return 0\n', '  local fl_k="" fl_v="" fl_n=0\n')],
 "R2 one: no -f check": [('  local on_l=""\n  [ -f "$1" ] || return 0\n', '  local on_l=""\n')],

 "R4 stat: no -f check": [('if [ -f "$pr_root/proc/stat" ]; then', 'if [ -e "$pr_root/proc/stat" ]; then')],
 "R5 all reads: no -t 1": [("read -r -t 1 -n 4096", "read -r -n 4096")],
 "R6 one: no -n 4096": [('  read -r -t 1 -n 4096 on_l <"$1"', '  read -r -t 1 on_l <"$1"')],
 "R7 field: no -n 4096": [('read -r -t 1 -n 4096 fl_k fl_v _', 'read -r -t 1 fl_k fl_v _')],
 "R8 field: no 64-line cap": [('while [ "$fl_n" -lt 64 ] &&', 'while')],
 "R9 psi: no 64-line cap": [('while [ "$ps_n" -lt 64 ] &&', 'while')],
 "R10 num: unbounded digits": [("^[0-9]{1,20}$", "^[0-9]+$")],
 "R11 stat: 19-digit fields": [("^[0-9]{1,18}$", "^[0-9]{1,19}$")],
 "R12 stat: no 10#": [("pr_sum=$((pr_sum + 10#$pr_v))", "pr_sum=$((pr_sum + pr_v))")],
 "R13 steal raw": [('pr_steal="$((10#$pr_st))"', 'pr_steal="$pr_st"')],
 "R14 total not via num": [('"$(num "$pr_total")"', '"$pr_total"')],
 "R15 nproc OMP not cleared": [("OMP_NUM_THREADS='' OMP_THREAD_LIMIT='' nproc", "nproc")],
 "R16 nproc THREAD_LIMIT kept": [("OMP_NUM_THREADS='' OMP_THREAD_LIMIT='' nproc", "OMP_NUM_THREADS='' nproc")],
 "R17 root from env (P7)": [('pr_root="${2:-/}"', 'pr_root="${2:-${ROOT:-/}}"')],
 "R18 cg from env": [('cg="$pr_root/sys/fs/cgroup"', 'cg="${CG:-$pr_root/sys/fs/cgroup}"')],
 "R19 field: 64 -> 640": [('"$fl_n" -lt 64', '"$fl_n" -lt 640')],

}

    for a, b in subs:
        assert text.count(a) >= 1, (name, a)

    d = out / name.split()[0]
    shutil.copytree(base, d)

    (d / "MUTANT").write_text(name + "\n")
