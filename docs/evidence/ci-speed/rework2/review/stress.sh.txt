#!/usr/bin/env bash
# stress.sh <busy loops> <repeats> <test id>...
busy=$1; n=$2; shift 2
for i in $(seq 1 "$busy"); do /opt/venv/bin/python -c 'while 1: pass' & done
for t in "$@"; do
  pass=0; fail=0; tmax=0
  for i in $(seq 1 "$n"); do
    s=$SECONDS
    if out="$(/opt/venv/bin/python -m unittest "$t" 2>&1)"; then pass=$((pass+1)); else fail=$((fail+1)); echo "--- $t run $i failed"; echo "$out" | grep -E 'Error|FAIL' | head -5; fi
    d=$((SECONDS-s)); [ $d -gt $tmax ] && tmax=$d
  done
  echo "nproc=$(nproc) busy=$busy $t: pass=$pass fail=$fail max_wall=${tmax}s"
done
kill $(jobs -p) 2>/dev/null
