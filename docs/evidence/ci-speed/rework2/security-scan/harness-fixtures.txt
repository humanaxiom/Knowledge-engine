import os, signal, subprocess, sys, time, unittest
class Ign(unittest.TestCase):
    def test_ign(self):
        signal.signal(signal.SIGTERM, signal.SIG_IGN)
        signal.signal(signal.SIGINT, signal.SIG_IGN)
        # grandchild in the same process group, also ignoring TERM/INT
        subprocess.Popen([sys.executable, '-c', 'import signal, time; signal.signal(signal.SIGTERM, signal.SIG_IGN); time.sleep(120)  # GRANDCHILD'])
        time.sleep(120)
