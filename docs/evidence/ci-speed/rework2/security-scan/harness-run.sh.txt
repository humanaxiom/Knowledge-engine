#!/usr/bin/env bash
# run.sh <name> <fixture dir> <cmd...>
S="$(cd "$(dirname "$0")/.." && pwd)"
name=$1; fx=$2; shift 2
MSYS_NO_PATHCONV=1 docker run --rm --name "ke-sec-r2-$name" --label ke.project=ke-sec-r2 \
  --user 10001:10001 --network none --cap-drop ALL --security-opt no-new-privileges:true \
  --tmpfs /tmp:exec,mode=1777 -e MYPY_CACHE_DIR=/tmp/mypy -e PYTHONDONTWRITEBYTECODE=1 \
  -e GIT_OPTIONAL_LOCKS=0 -v "$(cygpath -w "$S/fx/$fx"):/work:ro" -v "$(cygpath -w "$S/h"):/h:ro" -w /work \
  ke-gates:wt-33a132fbfdd7 "$@" > "$S/out/$name.txt" 2>&1
echo "rc=$?" >> "$S/out/$name.txt"
cat "$S/out/$name.txt"
