#!/usr/bin/env bash
# stress-inner.sh <busy loops> <repeats> <test id>
busy=$1; n=$2; t=$3
for i in $(seq 1 "$busy"); do /opt/venv/bin/python -c 'while 1: pass' & done
echo "nproc=$(nproc) busy=$busy repeats=$n"
pass=0; fail=0
for i in $(seq 1 "$n"); do
  if out="$(/opt/venv/bin/python -m unittest "$t" 2>&1)"; then pass=$((pass+1)); else fail=$((fail+1)); echo "--- run $i failed"; echo "$out" | grep -E 'Error|FAIL' | head -5; fi
done
echo "result: pass=$pass fail=$fail"
kill $(jobs -p) 2>/dev/null
