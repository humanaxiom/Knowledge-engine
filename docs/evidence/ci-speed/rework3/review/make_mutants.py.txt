 "T4_label_tests": ("scripts/test.sh", "cov_run toplevel -v", "cov_run tests -v"),

 "T6_sub_fail_ignored": ("scripts/test.sh", 'cov_run "$label" discover -s "$d" -t . -v || return $?', 'cov_run "$label" discover -s "$d" -t . -v || true'),

 "H1_rusage_self": ("tests/test_block_main_commit_hook.py", "resource.getrusage(resource.RUSAGE_CHILDREN)", "resource.getrusage(resource.RUSAGE_SELF)"),
 "H2_no_timeout": ("tests/test_block_main_commit_hook.py", "subprocess.run(args, timeout=HANG_GUARD_S, **kwargs)", "subprocess.run(args, **kwargs)"),
}
for name, (rel, old, new) in M.items():
    d = S / "mut" / name
    if d.exists(): shutil.rmtree(d)
    shutil.copytree(base, d)
    p = d / rel
    t = p.read_text(encoding="utf-8")
    n = t.count(old)
    assert n >= 1, (name, "pattern not found")
    p.write_text(t.replace(old, new), encoding="utf-8", newline="\n")
    print(name, "replaced", n)
