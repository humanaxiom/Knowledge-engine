        local spid; spid=$(cat "$PIDDIR/stage.pid" 2>/dev/null || echo none)

          group)  kill -s "$sig" -- "-$vpid" ;;
          verify) kill -s "$sig" "$vpid" ;;
          stage)  kill -s "$sig" "$spid" ;;
        esac
      fi

      sleep 3.5  # let any orphaned stage finish its log line

      outer_pass=$(grep -c 'verify full: PASS (' /tmp/out)

      printf '%-44s rc=%-3s outerPASS=%s innerPASS=%s last=[%s]\n      stages=%s\n' "$label" "$rc" "$outer_pass" "$inner_pass" "$last" "$(cat "$STAGE_LOG")"
    }
    echo "## control"
    run_case "no signal"                              none none  none   scanlike
    for phase in scan smoke; do
      mode=scanlike; [ "$phase" = smoke ] && mode=smokelike
      echo "## during $phase (stage stub mode: $mode)"
      for sig in INT TERM HUP QUIT USR1; do
        for target in group verify stage; do
          run_case "$sig to $target in $phase" "$phase" "$sig" "$target" "$mode"

        run_case "$sig to group in $phase (swallow0)" "$phase" "$sig" group swallow0
        run_case "$sig to stage in $phase (swallow0)" "$phase" "$sig" stage swallow0
      done
    done
