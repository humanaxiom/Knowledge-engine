def run(root: Path, tests: list[str]) -> tuple[int, str]:
    done = subprocess.run([sys.executable, "-m", "unittest", *tests], cwd=root, capture_output=True, text=True,
                          timeout=600)
    return done.returncode, done.stdout + done.stderr


def main() -> int:
    shutil.rmtree(BASE, True)
    shutil.copytree(SRC, BASE, ignore=shutil.ignore_patterns(".git", ".venv", "__pycache__", "_mutants.py"))
    all_tests = ["tests.test_image_scan", BIND]
    code, out = run(BASE, all_tests)
    print(f"GREEN copy: {'PASS' if code == 0 else 'FAIL'} ({out.strip().splitlines()[-1]})")
    if code != 0:
        print(out[-4000:])
        return 1
    killed = 0
    only = set(sys.argv[1:])
    for name, (tests, edits) in MUTANTS.items():
        if only and name not in only:
            continue
        shutil.rmtree(MUT, True)
        shutil.copytree(BASE, MUT)
        for rel, old, new in edits:
            path = MUT / rel
            if old is None:
                path.unlink()
                continue
            text = path.read_text(encoding="utf-8")
            assert old in text, f"{name}: anchor missing in {rel}: {old!r}"
            path.write_text(text.replace(old, new, 1), encoding="utf-8", newline="\n")
        code, out = run(MUT, tests)
        reason = ""
        for line in out.splitlines():
            if line.startswith("AssertionError"):
                reason = line[:150]
                break
        verdict = "KILLED" if code != 0 else "SURVIVED"
        killed += code != 0
        print(f"{verdict:8} {name:30} by {', '.join(t.rsplit('.', 1)[1] for t in tests)}  {reason}")
    total = len(only) if only else len(MUTANTS)
    print(f"{killed}/{total} mutants killed")
    return 0 if killed == total else 2


if __name__ == "__main__":
    sys.exit(main())
