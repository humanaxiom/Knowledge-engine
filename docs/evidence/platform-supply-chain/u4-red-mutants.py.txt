def run(root: Path, tests: list[str]) -> tuple[int, str]:
    d = subprocess.run([sys.executable, "-m", "unittest", *tests], cwd=root, capture_output=True, text=True,
                       timeout=1800)
    return d.returncode, d.stdout + d.stderr


def killers(out: str) -> str:
    names = sorted({m.group(1) for m in re.finditer(r"^(?:FAIL|ERROR): (\w+)", out, re.M)})
    first = next((l.strip()[:150] for l in out.splitlines() if l.startswith("AssertionError")), "")
    return f"by {', '.join(names)}  {first}" if names else ""


def mutate(base: Path, edits) -> None:
    shutil.rmtree(MUT, True)
    shutil.copytree(base, MUT, symlinks=True)
    for rel, old, new in edits:
        p = MUT / rel
        t = p.read_text(encoding="utf-8")
        assert old in t, f"anchor missing in {rel}: {old!r}"
        p.write_text(t.replace(old, new, 1), encoding="utf-8", newline="\n")
        if rel.endswith(".sh"):
            p.chmod(0o755)


def phase(title: str, base: Path, mutants: dict, controls: dict | None = None) -> int:
    print(f"\n## {title}", flush=True)
    tests = sorted({t for _, ts in mutants.values() for t in ts})
    code, out = run(base, tests)
    print(f"UNMUTATED ({len(tests)} tests): {'PASS' if code == 0 else 'FAIL'} ({out.strip().splitlines()[-1]})",
          flush=True)
    if code:
        print(out[-6000:])
        return 1
    killed = 0
    for name, (edits, targets) in mutants.items():
        mutate(base, edits)
        code, out = run(MUT, targets)
        killed += bool(code)
        print(f"{'KILLED' if code else 'SURVIVED':8} {name:40} {killers(out)}", flush=True)
    print(f"{killed}/{len(mutants)} killed", flush=True)
    rc = 0 if killed == len(mutants) else 1
    for name, (edits, targets) in (controls or {}).items():
        mutate(base, edits)
        code, out = run(MUT, targets)
        print(f"{'CONTROL-PASS' if code == 0 else 'CONTROL-FAIL'} {name:36} (must pass: an allowed change) "
              f"{killers(out)}", flush=True)
        rc |= bool(code)
    return rc


def main() -> int:
    shutil.rmtree(BASE, True)
    shutil.copytree(SRC, BASE, symlinks=True,
                    ignore=shutil.ignore_patterns(".git", ".venv", "__pycache__", "dist", "node_modules",
                                                  ".mypy_cache", ".coverage*"))
    shutil.rmtree(GREEN, True)
    shutil.copytree(BASE, GREEN, symlinks=True)
    for rel, src in ((V, "verify.green.sh"), (S, "scan.green.sh"), (A, "app.green.Dockerfile")):
        shutil.copyfile(f"/scratch/{src}", GREEN / rel)
        if rel.endswith(".sh"):
            (GREEN / rel).chmod(0o755)
    print("## RED on today's tree (each must fail):", flush=True)
    for t in RED:
        code, out = run(BASE, [t])
        print(f"{'RED' if code else 'NOT-RED'}  {t.rsplit('.', 1)[1]}  {killers(out)}", flush=True)
    rc = phase("Phase A: guard mutants on TODAY's tree", BASE, TODAY, CONTROLS)
    sc = subprocess.run(["shellcheck", "-x", V, S], cwd=GREEN, capture_output=True, text=True)
    print(f"\n## Phase B shellcheck -x on the green verify.sh + scan.sh: exit {sc.returncode} "
          f"{(sc.stdout + sc.stderr).strip() or '(no output)'}", flush=True)
    rc |= bool(sc.returncode)
    mods = ["tests.test_image_scan", "tests.test_stack_scripts", "tests.test_stack_model_keys",
            "tests.test_harness_port"]
    code, out = run(GREEN, mods)
    print(f"## Phase B satisfiability: {' '.join(mods)} on the green copy: "
          f"{'PASS' if code == 0 else 'FAIL'} ({out.strip().splitlines()[-1]}; "
          f"{[l for l in out.splitlines() if l.startswith('Ran ')]})", flush=True)
    if code:
        print(out[-8000:])
        rc = 1
    rc |= phase("Phase B: RED-test mutants on the green copy", GREEN, GREEN_MUT, GREEN_CONTROLS)
    rc |= phase("Phase B': guard mutants re-run on the green copy", GREEN, GREEN_GUARDS, CONTROLS)
    return rc


if __name__ == "__main__":
    sys.exit(main())
