    Step 'Get-NetTCPConnection -State Listen -LocalPort 47100,47101 (host listeners)' { Get-NetTCPConnection -State Listen -LocalPort 47100, 47101 -ErrorAction SilentlyContinue | ForEach-Object { '{0}:{1}' -f $_.LocalAddress, $_.LocalPort } | Sort-Object -Unique }
    Step 'TCP connect <this host non-loopback IPv4, not recorded>:47100 within 3 s (must be False)' {
        $ip = (Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.IPAddress -notlike '127.*' -and $_.IPAddress -notlike '169.254.*' -and $_.InterfaceAlias -notlike 'vEthernet*' } | Select-Object -First 1).IPAddress
        $c = [Net.Sockets.TcpClient]::new()
        $ok = $false
        try { $ok = $c.ConnectAsync($ip, 47100).Wait(3000) -and $c.Connected } catch { $ok = $false } finally { $c.Dispose() }
        "connected: $ok"
    }

    Section '5. Safety shim verification (before the refusal case)'
    $shimLog = Join-Path $scratch 'shim.log'
    if (Test-Path $shimLog) { Remove-Item $shimLog }
    $env:KE_SHIM_LOG = $shimLog
    $realPath = $env:PATH
    $env:PATH = "$shimDir;$realPath"
    Step 'shim: Get-Command docker (resolves to the shim)' { (Get-Command docker).Source -replace [regex]::Escape($scratch), '<scratch>' }
    Step "shim: docker version --format '{{.Server.Version}}' (passes through)" { docker version --format '{{.Server.Version}}' }
    Step 'shim: docker compose up --help (must be intercepted, exit 99; harmless if not)' { docker compose up --help | Select-Object -First 2 }
    Step 'shim call log' { Get-Content $shimLog }
    Remove-Item $shimLog

    Section '5a. Refusal: KE_BIND_ADDRESS=0.0.0.0 and KE_ENV=prod in .env (launch.ps1 under the shim)'
    (Get-Content .env) -replace '^KE_BIND_ADDRESS=.*$', 'KE_BIND_ADDRESS=0.0.0.0' -replace '^KE_ENV=.*$', 'KE_ENV=prod' | Set-Content .env
    Step 'diff .env.example .env' { EnvDiff }
    Step 'pwsh -NoProfile -File scripts\launch.ps1 (under the shim)' { pwsh -NoProfile -File scripts\launch.ps1 }
    Step 'shim call log (no compose up)' { Get-Content $shimLog; ''; "compose up reached: $([bool](Get-Content $shimLog | Where-Object { $_ -match '\bcompose\b' -and $_ -match '\bup\b' }))" }
    $env:PATH = $realPath
    Remove-Item Env:KE_SHIM_LOG
    Step 'project ports (unchanged, 127.0.0.1 only)' { ProjectPorts }
    Step 'Restore .env from <scratch>\env.bak; diff .env.example .env' { Copy-Item $envBak .env -Force; EnvDiff }
}
