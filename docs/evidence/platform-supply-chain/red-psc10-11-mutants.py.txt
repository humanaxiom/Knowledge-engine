def run(path, names):
    cli.GATES_DOCKERFILE = path
    lp.GATES_DOCKERFILE = path
    suite = unittest.TestSuite()
    for n in names:
        suite.addTests(unittest.TestLoader().loadTestsFromName(n))
    buf = io.StringIO()
    res = unittest.TextTestRunner(stream=buf, verbosity=0).run(suite)
    return res, buf.getvalue()


def desc(t):
    case = getattr(t, "test_case", t)
    sub = t._subDescription() if hasattr(t, "_subDescription") else ""
    return f"{case._testMethodName} {sub}".strip()


OUT.mkdir(parents=True, exist_ok=True)
g = OUT / "green.Dockerfile"
g.write_text(green, encoding="utf-8", newline="\n")

      f"ran={res.testsRun} failures={len(res.failures)} errors={len(res.errors)} skipped={len(res.skipped)}")
if res.failures or res.errors:

survived = 0
for name, body in mutants.items():
    assert body != green, name
    p = OUT / f"{name}.Dockerfile"
    p.write_text(body, encoding="utf-8", newline="\n")
    res, text = run(p, NEW)
    failed = sorted({desc(t) for t, _ in res.failures + res.errors})
    survived += not failed
    print(f"{name}: {'KILLED' if failed else 'SURVIVED'}")
    for f in failed:
        print(f"    {f}")
print(f"mutants: {len(mutants)}, survived: {survived}")
