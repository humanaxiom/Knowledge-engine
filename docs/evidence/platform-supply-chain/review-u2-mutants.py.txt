def run(root, tests):
    d = subprocess.run([sys.executable, "-m", "unittest", *tests], cwd=root, capture_output=True, text=True, timeout=900)
    return d.returncode, d.stdout + d.stderr

def main():
    shutil.rmtree(BASE, True)
    shutil.copytree(SRC, BASE, ignore=shutil.ignore_patterns(".git", ".venv", "__pycache__", "_review_mutants.py"))
    code, out = run(BASE, ALL)
    print(f"GREEN copy: {'PASS' if code == 0 else 'FAIL'} ({out.strip().splitlines()[-1]})", flush=True)
    if code:
        print(out[-4000:]); return 1
    for name, edits in MUTANTS.items():
        shutil.rmtree(MUT, True); shutil.copytree(BASE, MUT)
        for rel, old, new in edits:
            p = MUT / rel; t = p.read_text(encoding="utf-8")
            assert old in t, f"{name}: anchor missing in {rel}: {old!r}"
            p.write_text(t.replace(old, new, 1), encoding="utf-8", newline="\n")
        code, out = run(MUT, ALL)
        reason = next((l[:140] for l in out.splitlines() if l.startswith(("AssertionError", "FAIL:"))), "")
        print(f"{'KILLED' if code else 'SURVIVED':8} {name:30} {reason}", flush=True)
    return 0

if __name__ == "__main__":
    sys.exit(main())
