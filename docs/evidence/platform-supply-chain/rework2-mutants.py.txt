def run(mod, text, tag):
    p = OUT / (tag.split()[0] + ".Dockerfile"); p.write_text(text, encoding="utf-8")
    mod.GATES_DOCKERFILE = p; glp.GATES_DOCKERFILE = p
    suite = unittest.TestSuite([unittest.defaultTestLoader.loadTestsFromModule(mod),
                                unittest.defaultTestLoader.loadTestsFromModule(glp)])
    res = unittest.TextTestResult(io.StringIO(), False, 0); suite.run(res)
    fails = []
    for t, _ in res.failures + res.errors:
        s = str(t); i = s.find("check=")
        fails.append(s[i:].rstrip(")") if i >= 0 else s.split(" ")[0])
    return res.testsRun, fails

def verdict(fails): return f"KILLED ({len(fails)})" if fails else "SURVIVED"

        print(f"| {setname} | {tag} | {verdict(fb)} | {verdict(fa)} | {'; '.join(fa) or '-'} |")
