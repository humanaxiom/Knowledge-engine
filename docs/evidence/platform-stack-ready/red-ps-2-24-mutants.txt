def fresh_copy() -> None:
    if SCRATCH.exists():
        import os
        import stat

        def unlock(func, p, _exc):  # git objects are read-only on Windows
            os.chmod(p, stat.S_IWRITE)
            func(p)

        shutil.rmtree(SCRATCH, onerror=unlock)
    shutil.copytree(REPO, SCRATCH, ignore=shutil.ignore_patterns(
        ".git", ".venv", "__pycache__", "docs", "*.pyc"))
    # The repository scan lists files with git: give the copy a repository
    # (index only, the same tracked + untracked set as the working tree).
    subprocess.run(["git", "init", "-q"], cwd=SCRATCH, check=True)
    subprocess.run(["git", "-c", "core.autocrlf=false", "add", "-A"], cwd=SCRATCH, check=True)
    subprocess.run(["git", "config", "--global", "--get", "safe.directory"], capture_output=True)


def run(tests: list[str]) -> tuple[int, str]:
    cmd = ["docker", "run", "--rm", "--network", "none", "-v", f"{SCRATCH}:/work", "-w", "/work",
           "ke-gates", "uv", "run", "--locked", "python", "-m", "unittest", "-v", *tests]
    p = subprocess.run(cmd, capture_output=True, text=True, env={**__import__("os").environ,
                                                                  "MSYS_NO_PATHCONV": "1"})
    return p.returncode, p.stdout + p.stderr


def main() -> None:
    only = sys.argv[2:]
    all_tests = sorted({t for *_, ts in M for t in ts})
    fresh_copy()
    t0 = time.time()
    rc, out = run(all_tests)
    tail = [l for l in out.splitlines() if l.startswith(("Ran ", "OK", "FAILED"))]
    print(f"# Baseline (unmutated working tree): rc={rc} {tail} ({time.time() - t0:.0f}s)")
    if rc != 0:
        print(out)
    for mid, rel, old, new, tests in M:
        if only and not any(o in mid for o in only):
            continue
        fresh_copy()
        path = SCRATCH / rel
        text = path.read_text(encoding="utf-8")
        if text.count(old) != 1:
            print(f"[NOT-APPLIED] {mid}: pattern found {text.count(old)} times in {rel}")
            continue
        path.write_text(text.replace(old, new), encoding="utf-8", newline="\n")
        t0 = time.time()
        rc, out = run(tests)
        verdict = "RED (mutant KILLED)" if rc != 0 else "GREEN (mutant SURVIVED)"
        print(f"[{verdict}] {mid} ({time.time() - t0:.0f}s)")
        for line in out.splitlines():
            if line.startswith(("FAIL:", "ERROR:", "AssertionError", "Ran ", "FAILED", "OK")):
                print("    " + line)


main()
