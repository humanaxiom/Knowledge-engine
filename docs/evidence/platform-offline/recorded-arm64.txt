#!/usr/bin/env bash
# Recorded-run shim on aria-gb10 (tester, platform-offline): a pass-through to
# /usr/bin/docker, except that with KE_PROOF_D1=1 (set only for the bundle
# step) the defect-D1 filter answers `compose ... config --images <svc>`.
set -u
if [ "${KE_PROOF_D1:-}" = 1 ]; then
  # shellcheck source=/dev/null
  . "$HOME/ke-proof-tools/d1-filter.inc"
  d1_filter /usr/bin/docker "$@"
  exit
fi
exec /usr/bin/docker "$@"
# DEVIATION (defect D1, recorded): `docker compose config --images <svc>` prints
# the images of <svc> AND of its depends_on services (Compose 5.1.4 and 5.2.0
# both do; the order varies), so bundle.sh stops with "service worker resolves
# to more than one image". To exercise the steps after the bundle, this filter
# answers exactly that call shape with the service's own image, read from
# `docker compose ... config --format json`. Every other call passes through.
d1_filter() { # d1_filter REAL_DOCKER ARGS...
  local real="$1"; shift
  local n=$#
  if [ "$n" -ge 4 ] && [ "$1" = compose ] && [ "${@: -3:1}" = config ] && [ "${@: -2:1}" = --images ]; then
    local svc="${!n}"
    case "$svc" in -*) ;; *)
      echo "d1-filter: compose config --images $svc -> own image only (recorded deviation)" >&2
      "$real" "${@:1:n-3}" config --format json \
        | python3 -I -c 'import json,sys; print(json.load(sys.stdin)["services"][sys.argv[1]]["image"])' "$svc"
      return
    esac
  fi
  "$real" "$@"
}
