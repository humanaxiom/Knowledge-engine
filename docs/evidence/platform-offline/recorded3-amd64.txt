#!/usr/bin/env bash
# Recorded-run shim (tester, platform-offline, round 3). WSL Ubuntu on this dev
# host has no Docker Desktop WSL integration, so the `docker` on its PATH is
# Docker Desktop's stub that refuses. This forwards to the Windows docker.exe
# (same daemon, desktop-linux context), translating an absolute Linux path
# argument that exists (or whose directory exists) to its Windows form with
# wslpath, and forwarding the COMPOSE_* / DOCKER_CONTEXT variables docker itself
# reads (WSLENV /w: WSL -> Windows). No D1 filter: every call passes through.
set -u
args=()
for a in "$@"; do
  case "$a" in
    /*)
      if [ -e "$a" ] || [ -d "$(dirname -- "$a")" ]; then a="$(wslpath -w "$a")"; fi
      ;;
  esac
  args+=("$a")
done
fwd=""
for v in COMPOSE_PROJECT_NAME COMPOSE_PROFILES COMPOSE_FILE DOCKER_CONTEXT; do
  if [ -n "${!v+x}" ]; then fwd="${fwd:+$fwd:}$v/w"; fi
done
export WSLENV="${WSLENV:+$WSLENV:}$fwd"
exec "/mnt/c/Program Files/Docker/Docker/resources/bin/docker.exe" ${args[@]+"${args[@]}"}
