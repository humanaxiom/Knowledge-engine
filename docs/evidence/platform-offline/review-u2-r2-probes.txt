## M-4 in-place rewrite probe (tests/scratch_m4.py in scratch only)
import json, unittest
from pathlib import Path
from tests.test_offline_bundle import InstallShTest, InstallPs1Test

class M4Rewrite(InstallShTest):
    """In-place rewrite (same inode, truncate) of the bundle's install.sh at docker info."""
    def test_rewrite_in_place(self):
        bundle, target = self._fresh()
        injected = self.tmp / "INJECTED"
        src = (bundle / "install.sh").read_text(encoding="utf-8")
        assert src.endswith(chr(101) + "xit" + chr(10)), src[-40:]
        evil = src[: -len("exit\n")] + f"touch '{injected}'; exit\n"
        hooks = [{"on": r"^info\b", "write": {str(bundle / "install.sh"): evil}}]
        result, calls, _ = self._install(bundle, target, FAKE_DOCKER_HOOKS=json.dumps(hooks))
        print("rc", result.returncode, "injected", injected.exists())
        self.assertFalse(injected.exists(), "M-4: in-place rewrite executed\n" + self._out(result))

class M4RewritePs1(InstallPs1Test):
    def test_rewrite_in_place(self):
        bundle, target = self._fresh()
        injected = self.tmp / "INJECTED"
        p = bundle / "install.ps1"
        src = p.read_text(encoding="utf-8")
        evil = src + f"\nNew-Item -ItemType File -Path '{injected}' | Out-Null\n"
        hooks = [{"on": r"^info\b", "write": {str(p): evil}}]
        result, calls, _ = self._install(bundle, target, FAKE_DOCKER_HOOKS=json.dumps(hooks))
        print("rc", result.returncode, "injected", injected.exists())
        self.assertFalse(injected.exists(), self._out(result))

def load_tests(loader, tests, pattern):
    s = unittest.TestSuite()
    s.addTest(M4Rewrite("test_rewrite_in_place"))
    s.addTest(M4RewritePs1("test_rewrite_in_place"))
    return s
