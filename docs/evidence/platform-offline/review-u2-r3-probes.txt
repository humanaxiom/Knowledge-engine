import os, unittest
from pathlib import Path
from tests.test_offline_bundle import InstallShTest, STAGING_PREFIX


class SwapToLink(InstallShTest):
    def test_swap_to_link(self):
        bundle, target = self._fresh()
        victim = self.tmp / "victim-home"
        victim.mkdir()
        victim.chmod(0o700)
        (victim / "id_ed25519").write_text("SECRET KEY\n", encoding="utf-8")
        (victim / "sub").mkdir()
        (victim / "sub" / "keep").write_text("keep\n", encoding="utf-8")
        racer = self.tmp / "racer.log"
        shimdir = self._shim("chmod", f"""REAL "$@" || exit $?
for a in "$@"; do
  case "$a" in
    */{STAGING_PREFIX}*)
      if [ -d "$a" ] && [ ! -L "$a" ]; then mv -- "$a" "$a.away" && ln -s -- '{victim}' "$a" && echo "swapped $a" >> '{racer}'; fi ;;
  esac
done
""")
        result, _, log = self._install(bundle, target, PATH=f"{shimdir}{os.pathsep}{self.path}")
        out = self._out(result)
        print("racer:", racer.read_text() if racer.exists() else "(none)")
        print("rc", result.returncode, "docker calls:", bool(log))
        print("victim after:", sorted(str(p.relative_to(victim)) for p in victim.rglob("*")))
        print(out[-600:])
        self.assertTrue((victim / "id_ed25519").exists(), "cleanup deleted the victim's files")
