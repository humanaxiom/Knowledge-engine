"""Apply one mutant per scratch copy (tester, unit-2 rework 3)."""
import re
from pathlib import Path

M = Path(r"<scratch>\mut")


def edit(name, rel, fn):
    p = M / name / rel
    s = p.read_bytes().decode("utf-8")
    t = fn(s)
    assert t != s, (name, rel)
    p.write_bytes(t.encode("utf-8"))


# fix1: the developer's expected fix, `main "$@"; exit` on one line.
edit("fix1", "scripts/install.sh", lambda s: s.replace('main "$@"\nexit\n', 'main "$@"; exit\n'))


# unwrap: the main wrapper removed, the trailing exit kept (body runs at top level).
def unwrap(s):
    head, _, rest = s.partition("main() {\n")
    body, _, tail = rest.partition('\n}\n\nmain "$@"\nexit\n')
    assert tail == "" and body, "unexpected install.sh shape"
    body = re.sub(r"(?m)^(\s*)local ", r"\1declare ", body)
    return head + body + "\nexit\n"


edit("unwrap", "scripts/install.sh", unwrap)
if __import__("sys").argv[1:] == ["unwrap"]:
    raise SystemExit(0)

# clshc: the marker check removed from the sh cleanup.
edit("clshc", "scripts/install.sh",
     lambda s: s.replace(' && [ -f "$target/$claim_marker" ]; then\n    rm -rf', '; then\n    rm -rf'))

# pspath: Assert-SafePath made a no-op.
edit("pspath", "scripts/install.ps1",
     lambda s: s.replace("function Assert-SafePath {\n", "function Assert-SafePath {\n    return\n"))

# atl: at_least always true.
edit("atl", "scripts/launch.sh",
     lambda s: s.replace("at_least() { # at_least MINIMUM VERSION", "at_least() { return 0; # at_least MINIMUM VERSION"))
print("ok")
