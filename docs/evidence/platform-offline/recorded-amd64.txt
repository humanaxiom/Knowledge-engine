#!/usr/bin/env bash
# Recorded-run shim (tester, platform-offline). WSL Ubuntu on this dev host has
# no Docker Desktop WSL integration, so the `docker` on its PATH is Docker
# Desktop's stub that refuses. This forwards to the Windows docker.exe (same
# daemon, desktop-linux context), translating an absolute Linux path argument
# that exists (or whose directory exists) to its Windows form with wslpath, and
# forwarding the COMPOSE_* / DOCKER_CONTEXT variables docker itself reads.
set -u
args=()
for a in "$@"; do
  case "$a" in
    /*)
      if [ -e "$a" ] || [ -d "$(dirname -- "$a")" ]; then a="$(wslpath -w "$a")"; fi
      ;;
  esac
  args+=("$a")
done
fwd=""
for v in COMPOSE_PROJECT_NAME COMPOSE_PROFILES COMPOSE_FILE DOCKER_CONTEXT; do
  if [ -n "${!v+x}" ]; then fwd="${fwd:+$fwd:}$v/u"; fi
done
export WSLENV="${WSLENV:+$WSLENV:}$fwd"
exec "/mnt/c/Program Files/Docker/Docker/resources/bin/docker.exe" ${args[@]+"${args[@]}"}

=== STEP 3c-shim-v2-source  start 2026-10-19T18:50:20Z  cwd <home>/ke-proof-src

# DEVIATION (defect D1, recorded): `docker compose config --images <svc>` prints
# the images of <svc> AND of its depends_on services (Compose 5.1.4 and 5.2.0
# both do; the order varies), so bundle.sh stops with "service worker resolves
# to more than one image". To exercise the steps after the bundle, this filter
# answers exactly that call shape with the service's own image, read from
# `docker compose ... config --format json`. Every other call passes through.
d1_filter() { # d1_filter REAL_DOCKER ARGS...
  local real="$1"; shift
  local n=$#
  if [ "$n" -ge 4 ] && [ "$1" = compose ] && [ "${@: -3:1}" = config ] && [ "${@: -2:1}" = --images ]; then
    local svc="${!n}"
    case "$svc" in -*) ;; *)
      echo "d1-filter: compose config --images $svc -> own image only (recorded deviation)" >&2
      "$real" "${@:1:n-3}" config --format json \
        | python3 -I -c 'import json,sys; print(json.load(sys.stdin)["services"][sys.argv[1]]["image"])' "$svc"
      return
    esac
  fi
  "$real" "$@"
}
