class ProbeSh(_Probes, InstallShTest):
    def _mktemp_shim(self, body):
        return self._shim("mktemp", f"""case " $* " in
  *" -d "*{STAGING_PREFIX}*)
    out="$(REAL "$@")" || exit $?
{body}
    printf '%s\n' "$out"
    exit 0 ;;
esac
exec REAL "$@"
""")

    def _victim(self, files):
        v = self.tmp / f"home{self.n}" / ".ssh"
        v.mkdir(parents=True)
        if files:
            (v / "id_ed25519").write_text("PRIVATE KEY\n", encoding="utf-8")
            (v / "sub").mkdir()
            (v / "sub" / "keep").write_text("keep\n", encoding="utf-8")
        v.chmod(0o700)
        return v

    def _listing(self, v):
        return sorted(str(p.relative_to(v)) for p in v.rglob("*"))

    def test_probe_rb6_ssh_symlink(self):
        """RB6 re-run: staging name swapped for a link to a non-empty 0700 ~/.ssh."""
        bundle, target = self._fresh()
        v = self._victim(True)
        before = self._listing(v)
        shim = self._mktemp_shim(f"""    mv -- "$out" "$out.moved" && ln -s -- '{v}' "$out\"""")
        result, _, log = self._install(bundle, target, PATH=f"{shim}{os.pathsep}{self.path}")
        _report("rb6_ssh_symlink", result, f"victim before={before} after={self._listing(v)} mode={oct(stat.S_IMODE(v.stat().st_mode))} docker_ran={bool(log)}")

    def test_probe_rb6_empty_victim(self):
        """Staging name swapped for a link to an EMPTY operator-owned 0700 dir: only pwd -P refuses."""

        result, _, log = self._install(bundle, target, PATH=f"{shim}{os.pathsep}{self.path}")

    def test_probe_planted_in_staging(self):
