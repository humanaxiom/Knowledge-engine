## Driver (mutants.py)
"""Rework-1 mutation driver: run inside ke-gates, scratch copy at /work."""
import subprocess, sys, pathlib, time, json
W = pathlib.Path("/work")
SUITE = ["tests.test_stack_scripts", "tests.test_stack_entrypoints", "tests.test_stack_compose",
         "tests.test_stack_bind_address", "tests.test_stack_carry_ins", "tests.test_engine_tracing_guard",
         "tests.test_engine_telemetry", "tests.test_engine_startup", "tests.test_engine_hardcoded",
         "tests.test_engine_deploy_files", "tests.test_harness_followups", "tests.test_harness_port"]
M = json.loads(pathlib.Path("/mut/mutants.json").read_text())
only = sys.argv[1:]
for mid, rel, old, new in M:
    if only and mid.split(" ")[0] not in only:
        continue
    p = W / rel
    orig = p.read_bytes()
    text = orig.decode()
    if old not in text:
        print(f"[NOAPPLY] {mid}", flush=True); continue
    p.write_bytes(text.replace(old, new, 1).encode())
    t = time.time()
    try:
        r = subprocess.run(["uv", "run", "--locked", "python", "-m", "unittest", *SUITE], cwd=W,
                           capture_output=True, text=True, timeout=1800)
        fails = [l for l in r.stderr.splitlines() if l.startswith(("FAIL:", "ERROR:"))]
        status = "KILLED" if r.returncode else "SURVIVED"
        print(f"[{status}] {mid} ({time.time()-t:.0f}s)", flush=True)
        for l in sorted(set(fails))[:5]:
            print("    " + l, flush=True)
    finally:
        p.write_bytes(orig)
