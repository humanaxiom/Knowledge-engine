# argv: lockpath polpath dir mode -- swap fetch.json right after its lstat (mode: fifo|link|sparse|dir|devlink), then run sources-check
import sys, os
sys.path.insert(0, "/work"); os.chdir("/work")
from tests.test_sbom_bundle import load_sbom_check
S = load_sbom_check()
lock, pol, d, mode = sys.argv[1:5]
real = os.lstat
done = []
def sw(p, *a, **k):
    st = real(p, *a, **k)
    if str(p).endswith("fetch.json") and not done:
        done.append(1)
        good = open(p, "rb").read()
        os.unlink(p)
        if mode == "fifo": os.mkfifo(p)
        elif mode == "link":
            other = os.path.join(os.path.dirname(d), os.path.basename(d) + "-other.json")
            open(other, "wb").write(good); os.symlink(other, p)
        elif mode == "devlink": os.symlink("/dev/zero", p)
        elif mode == "sparse":
            fh = open(p, "wb"); fh.truncate(8 * 1024**3); fh.close()
        elif mode == "dir": os.mkdir(p)
    return st
S.os.lstat = sw
sys.exit(S.main(["sources-check", "--lock", lock, "--policy", pol, "--dir", d, "--arch", "amd64", "--set", "full"]))
