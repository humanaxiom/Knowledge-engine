import re, sys
SEP = "[/" + chr(92) * 2 + "]"  # a slash or a backslash
scr_tail = SEP.join(["AppData", "Local", "Temp", "claude", "c--repos-Knowledge-engine",
                     "d5694db3-7c12-4346-835e-30f12d337cee", "scratchpad"])
prof = "(?:<profile>|[Cc]:" + SEP + "Users" + SEP + "adam)"
ansi = re.compile("\x1b\\[[0-9;]*[A-Za-z]")
def red(t):
    t = ansi.sub("", t)
    t = re.sub(prof + SEP + scr_tail, "<scratch>", t)
    t = re.sub(prof, "<profile>", t)
    t = re.sub(r"S-1-5-21-\d+-\d+-\d+(?:-\d+)?", "<sid>", t)
    t = re.sub(r"(https://[^\s\"']*github[^\s\"'?]*)\?[^\s\"']+", r"\1?<query omitted>", t)
    return t
out = []
for src in sys.argv[2:]:
    with open(src, encoding="utf-8", errors="replace", newline="") as f:
        out.append(red(f.read().replace("\r\n", "\n")))
with open(sys.argv[1], "w", encoding="utf-8", newline="\n") as f:
    f.write("".join(out))
t = "".join(out)
print("lines", t.count("\n"), "leftover profile/SID hits:", len(re.findall(r"(?i)users[/\\]adam|S-1-5-21", t)))
