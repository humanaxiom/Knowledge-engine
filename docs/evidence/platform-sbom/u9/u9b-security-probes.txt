# Fragments of /repo/scripts/bundle.sh are extracted verbatim by line range (sed -n) and eval'd.
set -u
B=/repo/scripts/bundle.sh
frag() { sed -n "$1,$2p" "$B"; }
die() { echo "DIE: $*"; return 1; }
stage_die() { echo "STAGE_DIE ($1): $2"; return 1; }
OVERLAP="$(frag 155 170)"     # real_dir + overlap compare
OUTCHK="$(frag 122 138)"      # --out symlink/exists/empty + record checks
CLEAN="$(frag 433 455)"       # created=; bundle_cleanup()
grep -q '^real_dir()' <<< "$OVERLAP" || { echo "fragment drift (overlap)"; exit 9; }
grep -q '^bundle_cleanup() {' <<< "$CLEAN" || { echo "fragment drift (cleanup)"; exit 9; }
grep -q '^if \[ -L "\$out" \]' <<< "$OUTCHK" || { echo "fragment drift (out)"; exit 9; }

overlap() {  # $1 sources (as bundle.sh holds it), $2 out
  ( sources="$1"; out="$2"; stage_die() { echo "  REFUSED: $2"; exit 1; }
    eval "$OVERLAP"; echo "  PASSED overlap check: sources_real=$sources_real out_real=$out_real" )
}
W=$(mktemp -d); cd "$W"
echo "== P1 overlap check (bundle.sh l.155-170) =="

echo "P1c sources inside out via <nonexistent>/..:"; mkdir -p d/b-amd64/src; overlap "$W/d/b-amd64/src" "$W/d/new2/../b-amd64"
echo "P1d sources = / (--sources // after the single %/ strip):"; overlap "/" "$W/d/x-amd64"

echo "P1g out with a '.' segment:"; overlap "$W/d/src" "$W/d/src/./x-amd64"

echo "== P2 --out checks (l.122-138) + cleanup (l.433-455) with out=<nonexistent>/../victim-amd64 =="

( out="$W/d/ghost/../victim-amd64"; repo_root=/; sbom_work=""; wrote=0; record_made=0; project=p; sbom_user=1:1
  eval "$OUTCHK" || { echo "  out checks refused"; exit 1; }

  wrote=1; mkdir -p -- "$out/images"; echo "  mkdir -p wrote into: $(cd "$W/d/victim-amd64" && ls)"

echo "  run exit=$?"

  eval "$CLEAN"; trap bundle_cleanup EXIT; wrote=1; mkdir "$out"; rmdir "$out"; ln -s "$W/d/elsewhere2" "$out"; exit 1 )

if mknod S/zero c 1 5 2>/dev/null; then timeout 2 cp -P -- S/zero F/zero; echo "  char device /dev/zero: exit=$? copied $(stat -c %s F/zero) bytes in 2 s (unbounded)"; else echo "  mknod not permitted"; fi

    source "$W/shav.sh"; echo "  accepted ${#source_shas[@]} line(s): $(printf %q "$check_text")" )
done
echo "== done =="

  esac
  created=1
  echo "  -- $kind:"; eval "$CLAIM" || true

def _short(text: str, t: Path) -> str:
    lines = [ln for ln in text.splitlines() if ln.startswith(("bundle", "usage"))]
    return " | ".join(lines[-2:]).replace(str(t), "<t>")

    def _shim(self, on: str, do: str) -> str:
        """A docker ahead of the fake: on a call whose argv matches the glob `on`, run `do` first."""
        d = self.tmp / f"shim{self.n + 1}"
        d.mkdir()
        (d / "action.sh").write_text(do + "\n", encoding="utf-8", newline="\n")
        s = d / "docker"
        s.write_text("#!/bin/sh\n" + f'case "$*" in *"{on}"*) sh {d / "action.sh"} ;; esac\n'
                     + f'exec {self.tmp / "fakebin" / "docker"} "$@"\n', encoding="utf-8", newline="\n")
        s.chmod(0o755)
        return f"{d}{os.pathsep}{self.path}"

        beyond = [c[0] for c in calls if c[:1] != ["info"]]

              f"writes={len(self._writes(calls))} :: {_short(result.stdout + result.stderr, self.tmp)}"
              + (f" :: {extra}" if extra else ""), flush=True)

"""U9b rework-2 security probes (security, 2026-10-29). Not a repo test.

Runs the real scripts/bundle.sh with tests/fake_docker.py through the tests'
own _SourcesBase harness, inside ke-gates, --network none. Every symlink and
FIFO is made in the container's /tmp. A one-line docker shim ahead of the fake
runs one shell action when a call's argv contains a substring, then execs the
fake (as in rework 1). Each case prints one result line.
"""
from __future__ import annotations

import os
import stat
from pathlib import Path

from tests.test_bundle_sources import _SourcesBase, _snapshot

NL = chr(10)


def _short(text: str, t: Path) -> str:
    lines = [ln for ln in text.splitlines() if ln.startswith(("bundle", "usage"))]
    return " | ".join(lines[-2:]).replace(str(t), "<t>")


class Probe(_SourcesBase):

    def _shim(self, on: str, do: str) -> str:
        d = self.tmp / f"shim{self.n + 1}"
        d.mkdir()
        (d / "action.sh").write_text(do + NL, encoding="utf-8", newline=NL)
        s = d / "docker"
        body = ["#!/bin/sh",
                'case "$*" in *"' + on + '"*) sh ' + str(d / "action.sh") + " ;; esac",
                "exec " + str(self.tmp / "fakebin" / "docker") + ' "$@"', ""]
        s.write_text(NL.join(body), encoding="utf-8", newline=NL)
        s.chmod(0o755)
        return f"{d}{os.pathsep}{self.path}"

    def _line(self, case: str, result, calls, extra: str = "") -> None:
        beyond = [c[0] for c in calls if c[:1] != ["info"]]
        a4 = [ln for ln in result.stderr.splitlines() if ln.startswith("bundle: FAIL (stage ")]
        msg = (f"  {case}: exit={result.returncode} docker-beyond-info={len(beyond)} "
               f"writes={len(self._writes(calls))} A4-lines={len(a4)} :: "
               f"{_short(result.stdout + result.stderr, self.tmp)}")
        if extra:
            msg += " :: " + extra.replace(str(self.tmp), "<t>")
        print(msg, flush=True)

    def _side(self, parent: Path) -> str:
        rec = self._record(parent / "bundle-amd64")
        sb = [p.name for p in parent.iterdir() if p.name.startswith(".ke-sbom.")]
        return f"record={os.path.lexists(rec)} ke-sbom-left={sb}"

    @staticmethod
    def _state(out: Path) -> str:
        if out.is_symlink():
            return "link->" + os.readlink(out)
        if out.is_dir():
            names = sorted(os.listdir(out))
            body = {n: (out / n).read_text().strip() for n in names if n.endswith(".txt") and (out / n).is_file()}
            mf = out / "MANIFEST.sha256"
            inm = ("other.txt" in mf.read_text()) if mf.is_file() else False
            return f"dir entries={len(names)} txt={body} other.txt-in-manifest={inm}"
        if out.exists():
            return "file " + repr(out.read_text().strip())
        return "absent"
