def gl_calls(run):
    return [u for u in run.net.calls if host_of(u) == GITLAB]


def outcome_of(fn):
    o = io.StringIO()
    with contextlib.redirect_stdout(o):
        try:
            fn()
            res = "returned"
        except Exception as exc:  # noqa: BLE001
            res = "raised " + type(exc).__name__
    return res, o.getvalue()


class P1_NoFailOpen(_FetchCase):
    """(1) full fails closed without or with a bad APKBUILD; the sample still checks distfiles bytes."""

    def test_p1a_full_plan_without_apkbuild_fails_closed(self):
        b = one_per_kind()

        def drop(real):
            return lambda lock, arch, which: [(k, f) for k, f in real(lock, arch, which) if f["name"] != "APKBUILD"]
        run = self.run_fetch(b.text(), b.routes(), which="full",
                             patches=[lambda F: mock.patch.object(F.S, "fetch_plan", drop(F.S.fetch_plan))])
        print("\n  P1a full, APKBUILD dropped from plan: rc=%d fail=%r skipped_in_text=%s record=%s"
              % (run.rc, run.fail_lines, "skipped" in run.text, "fetch.json" in run.listing()))
        self.assertEqual(run.rc, 1)
        self.assertNotIn("skipped", run.text)
        self.assertNotIn("fetch.json", run.listing())

    def test_p1b_full_apkbuild_route_404_fails_closed(self):
        b = one_per_kind()
        routes = b.routes()
        apkbuild = next(f for f in b.components[APK_KEY]["files"] if f["name"] == "APKBUILD")
        routes[apkbuild["url"]] = (404, {}, b"")
        run = self.run_fetch(b.text(), routes, which="full")
        print("\n  P1b full, APKBUILD answers 404: rc=%d fail=%r gitlab_calls=%d"
              % (run.rc, run.fail_lines, len(gl_calls(run))))
        self.assertEqual(run.rc, 1)
        self.assertNotIn("fetch.json", run.listing())
        self.assertEqual(len(gl_calls(run)), 1)

    def _apk_with_sums(self, sums_fn):
        dist = ("readline-8.2.13.tar.gz", blob("readline tarball", 300))
        patch_ = ("fix-build.patch", blob("readline patch", 90))
        b = one_per_kind()
        a = LockBuilder()
        a.apk(APK_KEY, patches=[patch_], plain=[("readline.trigger", blob("readline trigger", 40))],
              distfiles=[dist], sums=sums_fn(dist, patch_))
        b.components[APK_KEY] = a.components[APK_KEY]
        b.bodies.update(a.bodies)
        return b

    def test_p1c_sha512sums_digest_differs(self):
        # The APKBUILD lists both names, but its sha512 for the distfile is not the lock value.
        b = self._apk_with_sums(lambda d, p: [(d[0], "e" * 128), (p[0], h("sha512", p[1]))])
        for which in ("full", "sample"):
            run = self.run_fetch(b.text(), b.routes(), which=which)
            print("\n  P1c sha512sums digest differs, %s: rc=%d fail=%r skip_lines=%d gitlab_calls=%d"
                  % (which, run.rc, run.fail_lines, run.out.count("name check skipped"), len(gl_calls(run))))
            if which == "full":
                self.assertEqual(run.rc, 1)
                self.assertIn("sha512sums names differ", run.fail_lines[0])
                self.assertNotIn("fetch.json", run.listing())
            else:  # the ruled A81 trade-off: the sample does not read the APKBUILD
                self.assertEqual(run.rc, 0)
                self.assertEqual(gl_calls(run), [])

    def test_p1d_full_sha512sums_names_differ_fails(self):

class P3_PartialNeverFull(_IndexCase):
    """(3) a partial key never passes a full bundle; only the apk sample key may be partial."""

    def _short(self, which, drop):
        real = self.S.fetch_plan
        return lambda lock, arch, w: [(k, f) for k, f in real(lock, arch, w) if not (w == which and drop(k, f))]

    def _index(self, which, listing):
        return self.build(which, listing=listing)

    def test_p3a_index_full_partial_refused(self):
        for label, dropf in (("apk GitLab files dropped", lambda k, f: k == APK_KEY and host_of(f["url"]) == GITLAB),
                             ("deb one file dropped", lambda k, f: k == DEB_KEY and f["name"].endswith(".debian.tar.xz"))):
            drop = self._short("full", dropf)
            listing = {f["sha256"]: ("file", f["size"]) for _, f in drop(self.b.lock(), ARCH, "full")}
            with mock.patch.object(self.S, "fetch_plan", side_effect=drop):
                idx, probs = self._index("full", listing)
            print("\n  P3a index full, %s: index produced=%s partial problems=%s"
                  % (label, idx is not None, [p for p in probs if "partial" in p]))
            self.assertIsNone(idx)
            self.assertTrue([p for p in probs if "partial in a full bundle" in p])

    def test_p3b_index_sample_non_apk_partial_refused(self):
        drop = self._short("sample", lambda k, f: k == DEB_KEY and f["name"].endswith(".debian.tar.xz"))
        listing = {f["sha256"]: ("file", f["size"]) for _, f in drop(self.b.lock(), ARCH, "sample")}
        with mock.patch.object(self.S, "fetch_plan", side_effect=drop):
            idx, probs = self._index("sample", listing)
        print("\n  P3b index sample, deb partial: index produced=%s problems=%s"
              % (idx is not None, [p for p in probs if "partial" in p]))
        self.assertIsNone(idx)

    def test_p3c_index_sample_listing_with_a_gitlab_file_refused(self):
        plan = self.S.fetch_plan(self.b.lock(), ARCH, "sample")
        listing = {f["sha256"]: ("file", f["size"]) for _, f in plan}
        extra = next(f for f in self.b.lock()["components"][APK_KEY]["files"] if host_of(f["url"]) == GITLAB)
        listing[extra["sha256"]] = ("file", extra["size"])
        idx, probs = self._index("sample", listing)
        print("\n  P3c index sample, --files also holds a GitLab file: index produced=%s problems=%s"
              % (idx is not None, probs[:2]))
        self.assertIsNone(idx)

    def test_p3d_index_sample_control(self):
        plan = self.S.fetch_plan(self.b.lock(), ARCH, "sample")
        listing = {f["sha256"]: ("file", f["size"]) for _, f in plan}
        idx, probs = self._index("sample", listing)
        st = {}
        for c in (idx or {}).get("components", []):
            st[c["status"]] = st.get(c["status"], 0) + 1
        print("\n  P3d control index sample: produced=%s statuses=%s problems=%s" % (idx is not None, st, probs[:2]))
        self.assertIsNotNone(idx)

    def _ix(self, which, key=APK_KEY):
        lock = self.b.lock()
        files = [dict(f, path=f"sources/files/{f['sha256']}" if (host_of(f["url"]) == DIST if key == APK_KEY else i == 0)
                      else None) for i, f in enumerate(lock["components"][key]["files"])]
        return {"format": 1, "set": which, "arch": ARCH, "git_sha": GIT_SHA, "lock_sha256": "0" * 64,
                "components": [{"key": key, "kind": key.split(":", 1)[0], "status": "partial",
                                "files": files, "packages": [{"image": "valkey", "spdxid": "SPDXRef-x"}]}],
                "unparsed": [], "unidentified": [], "unlicensed": {}}

    def test_p3e_notices_full_partial_refused(self):
        text, probs = self.S.source_section(self._ix("full"))
        print("\n  P3e notices, full index with a partial apk row: produced=%s problems=%s" % (text is not None, probs))
        self.assertIsNone(text)
        text, probs = self.S.source_section(self._ix("sample"))
        print("  P3e control notices, sample index with the apk partial row: produced=%s" % (text is not None))
        self.assertIsNotNone(text)

    def test_p3f_notices_sample_deb_partial(self):
        text, probs = self.S.source_section(self._ix("sample", DEB_KEY))
        print("\n  P3f notices, hand-made sample index with a deb partial row: produced=%s problems=%s"
              % (text is not None, probs))


class P4_NoticesHostile(_IndexCase):
    """(4) hostile key, URL or image in the partial row: refused, never escaped."""

    def _ix(self):
        lock = self.b.lock()
        files = [dict(f, path=f"sources/files/{f['sha256']}" if host_of(f["url"]) == DIST else None)
                 for f in lock["components"][APK_KEY]["files"]]
        return {"format": 1, "set": "sample", "arch": ARCH, "git_sha": GIT_SHA, "lock_sha256": "0" * 64,
                "components": [{"key": APK_KEY, "kind": "apk", "status": "partial", "files": files,
                                "packages": [{"image": "valkey", "spdxid": "SPDXRef-x"}]}],
                "unparsed": [], "unidentified": [], "unlicensed": {}}

    def test_p4a_hostile_values_refused(self):
        def url(v):
            return lambda c: c["files"][0].update(url=v)
        cases = {
            "url pipe": url("https://gitlab.alpinelinux.org/a|b"),
            "url backtick": url("https://gitlab.alpinelinux.org/a`b"),
            "url html tag": url("https://gitlab.alpinelinux.org/<script>alert(1)</script>"),
            "url backslash": url("https://gitlab.alpinelinux.org/a\b"),
            "url space": url("https://gitlab.alpinelinux.org/a b"),
            "url newline row": url("https://gitlab.alpinelinux.org/a\n| x | y | z |"),
            "url CR": url("https://gitlab.alpinelinux.org/a\rb"),
            "url RLO": url("https://gitlab.alpinelinux.org/\u202eb"),
            "url empty": url(""),
            "url list": url(["x"]),
            "url None": url(None),
            "key pipe": lambda c: c.update(key=APK_KEY + "|x"),
            "key backtick": lambda c: c.update(key=APK_KEY + "`x"),
            "key newline": lambda c: c.update(key=APK_KEY + "\n## x"),
            "key int": lambda c: c.update(key=7),
            "image md link": lambda c: c["packages"][0].update(image="[x](javascript:alert(1))"),
            "image star": lambda c: c["packages"][0].update(image="valkey*"),
            "image pipe": lambda c: c["packages"][0].update(image="a|b"),
            "no packages": lambda c: c.update(packages=[]),
            "files a string": lambda c: c.update(files="x"),
            "files missing": lambda c: c.pop("files"),
            "every file pathed": lambda c: [f.update(path="p") for f in c["files"]],
            "no file pathed": lambda c: [f.update(path=None) for f in c["files"]],
        }
        print()
        for name, change in cases.items():
            ix = self._ix()
            change(ix["components"][0])
            text, probs = self.S.source_section(ix)
            print("  P4a %-18s -> %s" % (name, "PRODUCED (not refused)" if text is not None
                                         else "refused: " + probs[0][:110]))
            self.assertIsNone(text, name)
            if text is None:
                self.assertTrue(all("|" not in (p.split(":", 1)[0]) for p in probs))

    def test_p4b_hostile_lock_values_never_reach_the_index(self):

class P6_GitLabClosed(_FetchCase):
    @staticmethod
    def closed(b):
        routes = b.routes()
        for u in list(routes):
            if host_of(u) == GITLAB:
                routes[u] = _refuse
        return routes

    def test_p6_sample_and_full(self):
        for name, b in (("one_per_kind", one_per_kind()), ("wide", wide())):
            run = self.run_fetch(b.text(), self.closed(b))
            print("\n  P6 %s sample, GitLab refusing: rc=%d gitlab_calls=%d total_calls=%d record=%s"
                  % (name, run.rc, len(gl_calls(run)), len(run.net.calls), "fetch.json" in run.listing()))
            self.assertEqual(run.rc, 0)
            self.assertEqual(gl_calls(run), [])
        b = one_per_kind()
        run = self.run_fetch(b.text(), self.closed(b), which="full",
                             patches=[lambda F: mock.patch.object(F.R, "_ACTIVE_RETRIES", [3])])
        print("  P6 full, GitLab refusing: rc=%d call hosts=%r sleeps=%r fail=%r"
              % (run.rc, [host_of(u) for u in run.net.calls], run.sleeps.calls, run.fail_lines))
        self.assertEqual(run.rc, 1)
        self.assertEqual(len(run.net.calls), 1)
