def sha(path):
    with open(path, "rb") as fh:
        return hashlib.sha256(fh.read()).hexdigest()


def apply(path, old, new, count):
    with open(path, encoding="utf-8") as fh:
        text = fh.read()
    n = text.count(old)
    if n != count:
        sys.exit(f"STOP: {path}: {old[:70]!r} matches {n} places, want {count}")
    with open(path, "w", encoding="utf-8", newline="\n") as fh:
        fh.write(text.replace(old, new))


def run(i):
    label, cand, edit, rows = ENTRIES[i]
    dst = f"/tmp/m{i}"
    shutil.rmtree(dst, ignore_errors=True)
    shutil.copytree("/work", dst, symlinks=True,
                    ignore=shutil.ignore_patterns(".git", "dist", ".venv", "__pycache__", "evidence", ".env"))
    if cand:
        for path, old, new, count in CAND:
            apply(os.path.join(dst, path), old, new, count)
    if edit:
        path, old, new, count = edit
        apply(os.path.join(dst, path), old, new, count)
    ids = [ROWS.get(r, r) for r in rows]
    t0 = time.time()
    p = subprocess.run([sys.executable, "-m", "unittest", *ids], cwd=dst, capture_output=True, text=True)
    out = p.stderr
    tail = [ln for ln in out.splitlines() if ln.startswith(("Ran ", "OK", "FAILED"))]
    failed = sorted({".".join(m.group(1).rsplit(".", 2)[-2:]) for m in re.finditer(r"^(?:FAIL|ERROR): \S+ \(([^)]+)\)", out, re.M)})
    names = {".".join(v.rsplit(".", 2)[-2:]): k for k, v in ROWS.items()}
    print(f"[{i}] {label}")
    print(f"     rows={rows}")
    print(f"     failed={sorted(names.get(f, f) for f in failed)} ({' '.join(tail)}; {time.time() - t0:.0f} s)")
    blocks = re.split(r"^={70}\n", out, flags=re.M)
    for b in blocks[1:]:
        head = b.splitlines()[0]
        why = [ln for ln in b.splitlines() if re.match(r"^[A-Za-z]*(Error|Exception)\b", ln)]
        print(f"    {head[:200]}")
        if why:
            print(f"      why: {why[-1][:300]}")
    sys.stdout.flush()
    shutil.rmtree(dst, ignore_errors=True)
