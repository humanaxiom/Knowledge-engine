-- probe script --
#!/usr/bin/env bash
# Reviewer probes for U9d rework 1: the events_start/events_stop shape with stand-in commands.
set -euo pipefail
cd /tmp
run() { # run NAME STREAMCMD LAUNCHCMD
  local name="$1" stream="$2" launch="$3"
  events_pid="" events_file=""
  events_file="$(mktemp /tmp/ke-offline-events.XXXXXX 2> /dev/null)" || { events_file=""; }
  t_start=$SECONDS
  # same shape as events_start: timeout 900 <cmd> > file 2>/dev/null </dev/null &
  timeout 900 bash -c "$stream" _ "$(date +%s)" > "$events_file" 2> /dev/null < /dev/null &
  events_pid=$!
  launch_rc=0
  bash -c "$launch" || launch_rc=$?
  events_state=none events_rc=0 events_out="" events_n=0
  if kill "$events_pid" 2> /dev/null; then events_state=stopped; else events_state=ended; fi
  wait "$events_pid" 2> /dev/null || events_rc=$?
  events_n="$(awk 'END { print NR }' "$events_file" 2> /dev/null)" || events_n=0
  events_out="$(head -n 100 -- "$events_file" 2> /dev/null)" || events_out=""
  rm -f -- "$events_file" || true
  left="$(find /tmp -maxdepth 1 -name "ke-offline-events.*" | wc -l)"
  orphans=0; for f in /proc/[0-9]*/cmdline; do c="$(tr "\0" " " < "$f" 2> /dev/null || true)"; case "$c" in "sleep 77"*|"timeout 900"*) orphans=$((orphans + 1));; esac; done
  echo "$name: launch_rc=$launch_rc state=$events_state rc=$events_rc n=$events_n out=[${events_out//$'\n'/|}] tmpfiles_left=$left orphans=$orphans"
}
# P-a: stream running (a child that dies by SIGTERM, like the docker CLI), launch fails
run "running+signal-death" 'echo since=$1; exec sleep 77' 'sleep 0.3; exit 7'
# P-b: stream that traps TERM and exits 143 (the fake)
run "running+trap143" 'trap "exit 143" TERM; echo since=$1; sleep 77 & wait' 'sleep 0.3; exit 7'
# P-c: stream ended early (exit 1) well before launch exits
run "ended-early" 'echo since=$1; exit 1' 'sleep 0.5; exit 7'
# P-d: launch killed by a signal (SIGKILL on itself)
run "launch-sigkill" 'exec sleep 77' 'kill -9 $$'
# P-e: launch passes
run "launch-pass" 'exec sleep 77' 'sleep 0.2; exit 0'
# P-f: 130 lines then hang
run "cap" 'for i in $(seq 1 130); do echo "$i die x exitCode=$i signal="; done; exec sleep 77' 'sleep 0.3; exit 7'
echo "--- expansion order: is \$(date) in the background command evaluated in the child?"
t0=$(date +%s%N)
timeout 5 echo "$(sleep 1; echo x)" > /dev/null &
t1=$(date +%s%N); wait $! || true
echo "parent continued after $(( (t1 - t0) / 1000000 )) ms (about 0 = expanded in the child; about 1000 = in the parent)"
echo "--- errexit with a failing substitution in the background command"
timeout 5 echo "$(exit 3)" > /dev/null & wait $! ; echo "client still running after a failing \$(...) in the bg args: rc=$?"
echo "--- stderr of the whole probe is captured separately"

Reading: on every path the temp file is gone (tmpfiles_left=0) and no `timeout 900`/stream child survives
(orphans=0). `kill` + `wait ... || events_rc=$?` never trips `set -e`: a stream killed by TERM (as the docker CLI
is; timeout re-raises it) gives rc 143 and no job notice on stderr; one that traps TERM gives 143 too; one that
ended early is `ended` with its own status (1). A launch killed by SIGKILL gives launch_rc=137 and events_stop still
runs. The only stderr line is bash's FOREGROUND job notice for the SIGKILLed launch stand-in (pre-existing
behaviour for any signalled launch.sh, not the stream). The cap case keeps lines 1..100 and counts n=130.
`$(date +%s)` in the background command's arguments is expanded in the forked child: the parent went on after
1 ms (with a 1 s substitution), and a failing substitution there does not stop the parent.
(The `bash -x` trace of the first attempt also shows `bash -c 'sleep 0.3; exit 7'` traced before `date +%s`.)

== R2: neutralise under bash 5.2 (patsub_replacement on) ==
'a ##[error]x' -> 'a #?[error]x'; '###[x' -> '##?[x'; '##[##[' -> '#?[#?['; '#&##[&' -> '#&#?[&' (no `&`
substitution: the replacement is quoted); '##' and '[##' unchanged; '::error::x' unchanged (every printed line
carries the `offline-client: ` prefix, so `::` is never at line start). No output of neutralise holds `##[`.
