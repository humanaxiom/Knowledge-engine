==== <scratch>/tools-relockfix/sbom_tsv.py
"""Tester helper (relock fixtures 2026-11-02): the tsv subcommand of the u3 sbom_evidence.py
(docs/evidence/platform-sbom/u3/windows-run.txt l.1982-1995), verbatim logic: purl ('-' if none),
versionInfo, licenseDeclared|licenseConcluded, document order, LF. argv: <spdx.json> <out.tsv>"""
import json, pathlib, sys

def tsv(path):
    doc = json.loads(pathlib.Path(path).read_text(encoding="utf-8"))
    out = []
    for p in doc.get("packages") or []:
        purl = next((r.get("referenceLocator") for r in p.get("externalRefs") or [] if r.get("referenceType") == "purl"), None) or "-"
        out.append(f"{purl}\t{p.get('versionInfo', '')}\t{p.get('licenseDeclared', '')}|{p.get('licenseConcluded', '')}")
    return "\n".join(out) + "\n"

pathlib.Path(sys.argv[2]).write_bytes(tsv(sys.argv[1]).encode("utf-8"))
