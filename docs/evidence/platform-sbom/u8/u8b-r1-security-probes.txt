# Part C: probes-links.sh
#!/usr/bin/env bash
# U8b rework 1 link probes, run inside ke-gates (network none, read-only root, uid 1000), with the probe fake docker.
set -u
W=/tmp/w; rm -rf "$W"; mkdir -p "$W/bin" "$W/repo"; cp -r /probe/repo/. "$W/repo/"; cp /probe/bin/docker "$W/bin/docker"; chmod +x "$W/bin/docker"
cd "$W"; export PATH="$W/bin:$PATH" PROBE_LOG="$W/docker.log" HOME=/tmp
B="bash $W/repo/scripts/bundle_fetch.sh"
run() { echo "### $1"; shift; : > "$PROBE_LOG"; "$@" > out.txt 2>&1; echo "rc=$?"; sed 's/^/  | /' out.txt; echo "  docker calls: $(grep -c '^CALL' "$PROBE_LOG")"; }
echo "uid=$(id -u) gid=$(id -g) bash=$BASH_VERSION"
mkdir tgt; ln -s tgt link
for o in link link/ link// link/// link/. link/./ link//. ; do run "L1 --out $o (link to an empty dir)" $B --set sample --out "$W/$o"; done
echo "  tgt entries after L1: $(ls -A tgt | wc -l); link still a link: $([ -L link ] && echo yes)"
mkdir o4; ln -s /etc/hostname o4/fetch.json; run "L4 --out holding a fetch.json symlink" $B --set sample --out "$W/o4"; echo "  o4/fetch.json still a link: $([ -L o4/fetch.json ] && echo yes)"
mkdir o5; ln -s /nonexistent o5/dangling; run "L5 --out holding only a dangling link" $B --set sample --out "$W/o5"
mkdir real; ln -s real plink; run "L6 --out plink/out (link as a middle component, out absent)" $B --set sample --out "$W/plink/out"; echo "  out landed in real/: $([ -d real/out ] && echo yes) (residual: middle components are not checked)"
mkdir o7; echo keep > o7/operator.txt; chmod 0300 o7; run "L7 unlistable non-empty --out (mode 0300)" $B --set sample --out "$W/o7"; chmod 0700 o7; echo "  o7 after: $(ls -A o7)"
cat > "$W/bin/docker-wrap" <<'WR'
#!/usr/bin/env bash
"$PROBE_REAL" "$@"; rc=$?
case " $* " in *" run "*" sources-fetch "*) [ "$rc" -eq 0 ] && mv -- "$KE_SOURCES_OUT/fetch.json" /tmp/w/moved.json && ln -s /tmp/w/moved.json "$KE_SOURCES_OUT/fetch.json" ;; esac
exit $rc
WR
mkdir "$W/bin2"; cp "$W/bin/docker-wrap" "$W/bin2/docker"; chmod +x "$W/bin2/docker"
PROBE_REAL="$W/bin/docker" PATH="$W/bin2:$PATH" run "L8 fetch.json a link after the client (--out created)" $B --set sample --out "$W/o8"; echo "  o8 exists after: $([ -e o8 ] || [ -L o8 ] && echo yes || echo no)"
