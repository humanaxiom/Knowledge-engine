"""Reviewer probe (U8b): random argument lists against scripts/bundle_fetch.sh on
fake docker, the way tests/test_bundle_fetch.py runs it. Read-only on the repo."""
import json, os, random, re, sys, unittest
sys.path.insert(0, "/work")
from tests.test_bundle_fetch import _FetchScriptBase

SEED = int(os.environ.get("PROBE_SEED", "20261028"))
FLAGS = ["--set", "--out", "--arch"]
JUNK = ["", "-", "--", "-e", "--set=sample", "--out=x", "--ARCH", "sample", "full", "amd64", "arm64", "x86_64",
        "partial", "SAMPLE", " sample", "sample ", "--evidence-dir", "-h", "--help", "*", "$(touch PWNED)",
        "a b", "a\nb", ";touch PWNED", "`touch PWNED`", "../x", "/", "//", "."]

def spec_valid(args):
    """U8-1 / A75 grammar: flags with a value; --set in {sample, full}; --out non-empty;
    --arch absent or in {amd64, arm64}; nothing else. Repeats: last wins (not in the spec)."""
    i, seen = 0, {}
    while i < len(args):
        a = args[i]
        if a not in FLAGS or i + 1 >= len(args):
            return False
        seen[a] = args[i + 1]
        i += 2
    if seen.get("--set") not in ("sample", "full"):
        return False
    if not seen.get("--out"):
        return False
    return seen.get("--arch", "") in ("", "amd64", "arm64")

class Probe(_FetchScriptBase):
    def test_random_args(self):
        rng = random.Random(SEED)
        stats = {"invalid": 0, "valid": 0, "explicit-empty-arch": 0}
        report = []
        for n in range(int(os.environ.get("PROBE_N", "120"))):
            out = str(self._out())
            toks = []
            for _ in range(rng.randint(0, 7)):
                r = rng.random()
                if r < 0.45:
                    toks.append(rng.choice(FLAGS))
                elif r < 0.6:
                    toks.append(out)
                else:
                    toks.append(rng.choice(JUNK))
            if rng.random() < 0.35:  # bias towards near-valid lists
                toks = ["--set", rng.choice(["sample", "full", "partial", ""]), "--out", rng.choice([out, ""]),
                        *rng.choice([[], ["--arch", rng.choice(["amd64", "arm64", "", "x86_64"])]])]
                rng.shuffle_pairs = None
            valid = spec_valid(toks)
            r, recs = self._run(*toks)
            pwned = [p for p in (self.repo / "PWNED", self.tmp / "PWNED") if p.exists()]
            self.assertEqual(pwned, [], f"injection via {toks!r}")
            line = f"{n:03d} valid={valid} rc={r.returncode} docker_calls={len(recs)} args={toks!r}"
            report.append(line)
            if not valid:
                stats["invalid"] += 1
                if "--arch" in toks and toks[toks.index("--arch") + 1:toks.index("--arch") + 2] == [""] and r.returncode != 2:
                    stats["explicit-empty-arch"] += 1
                    report.append("     ^ explicit --arch '' accepted (spec: another value exits 2)")
                    continue
                self.assertEqual(r.returncode, 2, line + "\n" + r.stderr)
                self.assertEqual(recs, [], line)
                self.assertTrue(any(l.startswith("usage: scripts/bundle_fetch.sh") for l in r.stderr.splitlines()), line)
                self.assertFalse(os.path.lexists(out), line)
            else:
                stats["valid"] += 1
                self.assertIn(r.returncode, (0, 1), line + "\n" + r.stderr[-2000:])
                self.assertNotIn("Traceback", r.stdout + r.stderr, line)
                if r.returncode == 1:
                    self.assertFalse(os.path.lexists(out) and os.listdir(out), line)
        print("\n".join(report))
        print("STATS", json.dumps(stats))

if __name__ == "__main__":
    unittest.main()
