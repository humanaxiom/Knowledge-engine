"""Summarise profrun.py JSON outputs. Usage: analyze.py OUTDIR"""
import json, sys, os, collections
d = sys.argv[1]
def load(name):
    p = os.path.join(d, name + ".json")
    if not os.path.exists(p): return None
    j = json.load(open(p))
    # tests/test_sources_fetch.py appeared untracked (another role's U8 RED tests)
    # during the later runs; it is not part of either commit, so it is excluded.
    j["records"] = [r for r in j["records"] if r["id"].split(".")[1] != "test_sources_fetch"]
    j["tests"] = len(j["records"])
    return j
runs = {k: load(k) for k in ("head-cov", "pre-cov", "head-nocov", "pre-nocov")}
def mod(tid): return tid.split(".")[1]
def cls(tid): return ".".join(tid.split(".")[1:3])
def full(r): return (r["wall"] + r["gap_wall"], r["cpu"] + r["gap_cpu"], r["child_cpu"] + r["gap_child_cpu"])
def per(run, key):
    agg = collections.defaultdict(lambda: [0, 0.0, 0.0, 0.0])
    for r in run["records"]:
        w, c, k = full(r); a = agg[key(r["id"])]
        a[0] += 1; a[1] += w; a[2] += c; a[3] += k
    return agg
def header(run, name):
    print(f"### {name}: {run['tests']} tests; process wall {run['total_wall']:.1f} s, own CPU {run['total_cpu']:.1f} s, "
          f"reaped-children CPU {run['total_child_cpu']:.1f} s; import/load {run['load_wall']:.1f} s wall / {run['load_cpu']:.1f} s CPU; "
          f"results {dict(collections.Counter(r['result'] for r in run['records']))}")
for name, run in runs.items():
    if run: header(run, name); print()
h = runs["head-cov"]
if h:
    print("## Slowest 30 tests, head-cov (the gate's own invocation, under coverage)\n")
    print("| # | test | wall s | own CPU s | children CPU s | fixture gap before s |")
    print("|---|---|---:|---:|---:|---:|")
    rs = sorted(h["records"], key=lambda r: -(r["wall"]))[:30]
    for i, r in enumerate(rs, 1):
        print(f"| {i} | `{r['id'][6:]}` | {r['wall']:.2f} | {r['cpu']:.2f} | {r['child_cpu']:.2f} | {r['gap_wall']:.2f} |")
    print("\n## Largest fixture gaps (setUpModule/setUpClass of the next class + tearDownClass of the previous), head-cov\n")
    print("| next test | gap wall s | gap own CPU s | gap children CPU s |")
    print("|---|---:|---:|---:|")
    for r in sorted(h["records"], key=lambda r: -r["gap_wall"])[:10]:
        print(f"| `{r['id'][6:]}` | {r['gap_wall']:.2f} | {r['gap_cpu']:.2f} | {r['gap_child_cpu']:.2f} |")
def modtable(a, b, an, bn):
    A = per(a, mod); B = per(b, mod) if b else {}
    keys = sorted(set(A) | set(B), key=lambda k: -(A.get(k, [0, 0, 0, 0])[1] - B.get(k, [0, 0, 0, 0])[1]) if b else -A[k][1])
    print(f"| module | {an} tests | {an} wall s | {an} own CPU s | {an} child CPU s |" + (f" {bn} tests | {bn} wall s | {bn} own+child CPU s | delta wall s | delta CPU s |" if b else ""))
    print("|---|---:|---:|---:|---:|" + ("---:|---:|---:|---:|---:|" if b else ""))
    tot = [0, 0, 0, 0, 0, 0, 0]
    for k in keys:
        x = A.get(k, [0, 0, 0, 0]); y = B.get(k, [0, 0, 0, 0])
        line = f"| {k} | {x[0]} | {x[1]:.1f} | {x[2]:.1f} | {x[3]:.1f} |"
        if b:
            line += f" {y[0]} | {y[1]:.1f} | {y[2]+y[3]:.1f} | {x[1]-y[1]:+.1f} | {(x[2]+x[3])-(y[2]+y[3]):+.1f} |"
        print(line)
        for i, v in enumerate(x): tot[i] += v
        if b:
            tot[4] += y[0]; tot[5] += y[1]; tot[6] += y[2] + y[3]
    line = f"| **total** | {tot[0]} | {tot[1]:.1f} | {tot[2]:.1f} | {tot[3]:.1f} |"
    if b: line += f" {tot[4]} | {tot[5]:.1f} | {tot[6]:.1f} | {tot[1]-tot[5]:+.1f} | {(tot[2]+tot[3])-tot[6]:+.1f} |"
    print(line)
if h and runs["pre-cov"]:
    print("\n## Per-module totals, head-cov vs pre-cov (86f7831), sorted by wall delta\n")
    modtable(h, runs["pre-cov"], "head", "pre")
if h and runs["head-nocov"]:
    print("\n## Coverage multiplier per module: head-cov vs head-nocov\n")
    modtable(h, runs["head-nocov"], "cov", "nocov")
if runs["pre-cov"] and runs["pre-nocov"]:
    print("\n## Coverage multiplier per module: pre-cov vs pre-nocov\n")
    modtable(runs["pre-cov"], runs["pre-nocov"], "cov", "nocov")
if h and runs["pre-cov"]:
    print("\n## Per-test delta for tests present in both (head-cov minus pre-cov), top 20 by wall increase\n")
    P = {r["id"]: r for r in runs["pre-cov"]["records"]}
    rows = [(r["wall"] - P[r["id"]]["wall"], r, P[r["id"]]) for r in h["records"] if r["id"] in P]
    print("| test | head wall s | pre wall s | delta s |")
    print("|---|---:|---:|---:|")
    for dlt, r, p in sorted(rows, key=lambda t: -t[0])[:20]:
        print(f"| `{r['id'][6:]}` | {r['wall']:.2f} | {p['wall']:.2f} | {dlt:+.2f} |")
    new = [r for r in h["records"] if r["id"] not in P]
    print(f"\nNew tests in head (not in pre): {len(new)}, wall {sum(full(r)[0] for r in new):.1f} s, own+child CPU {sum(full(r)[1]+full(r)[2] for r in new):.1f} s")
    gone = [p for p in runs['pre-cov']['records'] if p['id'] not in {r['id'] for r in h['records']}]
    print(f"Tests in pre gone/renamed in head: {len(gone)}")
    print("\n## Per-class totals, head-cov, top 25 by wall\n")
    C = per(h, cls); PC = per(runs["pre-cov"], cls)
    print("| class | tests | wall s | own CPU s | child CPU s | pre wall s |")
    print("|---|---:|---:|---:|---:|---:|")
    for k, v in sorted(C.items(), key=lambda kv: -kv[1][1])[:25]:
        print(f"| {k} | {v[0]} | {v[1]:.1f} | {v[2]:.1f} | {v[3]:.1f} | {PC.get(k, [0, 0])[1]:.1f} |")
