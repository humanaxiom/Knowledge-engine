# probe fake docker: logs argv (NUL-safe via %q) and the KE_/COMPOSE_ env; never contacts a daemon

  info) printf '%s\n' "${PROBE_ARCH-x86_64}"; exit 0 ;;
  image) [ -n "${PROBE_NO_IMAGE:-}" ] && exit 1; echo linux/amd64; exit 0 ;;
esac
if [ "$1" = compose ]; then
  case " $* " in
    *" config --images sbom "*) echo 'anchore/syft:v1.54.1@sha256:3eb5379ba7b409c3f4069b686110527af0c47df993fa5c10d13e7cf34f49b1aa'; exit 0 ;;

  esac
